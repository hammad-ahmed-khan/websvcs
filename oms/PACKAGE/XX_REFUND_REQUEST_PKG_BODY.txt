             --,P_PROCESSED_FLAG                VARCHAR2
             -- ,P_PROCESSED_ON                        DATE
             
             P_PROCESSED_BY

             -- ,P_ERR_MSG                       VARCHAR2
            ,

       P_TENDER_TYPE,
       P_TENDER_SUBTYPE,
       P_DECIMAL_ADJUSTMENT,
       P_REFUND_AMOUNT,
       P_COUNTRY,
       P_CURRENCY,
       P_REFUNDABLE,
       P_SADAD_BANKID,
       v_SADAD_UID,
       P_SADAD_SPTN,
       v_SADAD_REFUND_ID,

       P_PAYFORT_MERCHANT_REFERENCE,
       P_TASHEEL_WALLET_CIVILID,
       P_TASHEEL_WALLET_REFNUMBER,
       P_TASHEEL_CARDNO,

       P_REQ_ADDITIONAL_ATTRIBUTE2,

  procedure update_xrr_processing_status(p_Refund_Id                 number,
                                         p_processed_flag            varchar2,
                                         p_processed_by              varchar2,
                                         p_refund_status             varchar2,
                                         p_refund_date               date,
                                         p_err_code                  varchar2,
                                         p_err_msg                   varchar2,
                                         p_RES_ADDITIONAL_ATTRIBUTE1 varchar2,
                                         x_status_code               out varchar2,
                                         x_status_msg                out varchar2) is
    v_rec_count number;
    v_error     varchar2(2000);
    v_refund_id number;
  begin
  
    insert into xx_refund_request_param_log
      (P_REFUND_ID,
       P_PROCESSED_FLAG,
       P_PROCESSED_BY,
       P_REFUND_STATUS,
       P_REFUND_DATE,
       P_ERR_CODE,
       P_ERR_MSG,
       PARAM_SOURCE,
       CREATION_DATE)
    values
      (P_REFUND_ID,
       P_PROCESSED_FLAG,
       P_PROCESSED_BY,
       P_REFUND_STATUS,
       P_REFUND_DATE,
       P_ERR_CODE,
       P_ERR_MSG,
       'RMS',
       sysdate
       
       );
    commit;
    
    if p_Refund_Id = 0 then
     DBMS_LOCK.sleep(seconds => 3);
      select max(refund_id)
        into v_refund_id
        from XX_REFUND_REQUEST
       where RES_ADDITIONAL_ATTRIBUTE1 = p_RES_ADDITIONAL_ATTRIBUTE1;
    else
      v_refund_id := p_Refund_Id;
    end if;
  
    select count(*)
      into v_rec_count
      from XX_REFUND_REQUEST xrr
     where xrr.refund_id = v_refund_id;
  
    if v_rec_count = 1 then
    
      update XX_REFUND_REQUEST xrr
         set xrr.processed_flag            = p_processed_flag,
             xrr.refund_status             = p_refund_status,
             xrr.processed_on              = sysdate,
             xrr.res_additional_attribute1 = p_RES_ADDITIONAL_ATTRIBUTE1,
             xrr.refund_method = case
                                   when refund_channel in
                                        ('Benefit',
                                         'NBO',
                                         'SADAD',
                                         'Tasheel Wallet') then
                                    refund_channel
                                   when refund_channel = 'Payfort' and
                                        COUNTRY = 'OM' then
                                    'Payfort'
                                   when refund_channel = 'Payfort' and
                                        COUNTRY = 'SA' and
                                        tender_subtype in ('3000', '3010') then
                                    'Payfort - CC - V. M.'
                                   when refund_channel = 'Payfort' and
                                        COUNTRY = 'SA' and
                                        tender_subtype in ('3020') then
                                    'AMEX'
                                   when refund_channel = 'Payfort' and
                                        COUNTRY = 'SA' and
                                        tender_subtype in ('8000') then
                                    'Payfort - DC - MADA'
                                   when refund_channel = 'FSS' and
                                        COUNTRY = 'BH' then
                                    'AFS'
                                 end,
             xrr.manual_processing_flag = case
                                            when p_refund_status = 'Rejected' then
                                             'Y'
                                            else
                                             'N'
                                          end,
             xrr.processed_by              = p_processed_by,
             xrr.err_code                  = p_err_code,
             xrr.err_msg                   = p_err_msg,
             xrr.last_update_date          = sysdate,
             xrr.refund_date               = sysdate
       where xrr.refund_id = v_refund_id;
    
      if p_processed_flag = 'Y' and
         p_refund_status in ('Accepted', 'Processed') then
      
        update XX_REFUND_REQUEST xrr
           set xrr.refund_ref_number = case
                                         when xrr.refund_channel = 'SADAD' then
                                          to_char(xrr.sadad_refund_id)
                                         when xrr.refund_channel = 'Payfort' then
                                          xrr.payfort_merchant_reference || ' ' ||
                                          xrr.refund_id
                                         else
                                          xrr.cust_order_no
                                       end,
               xrr.refunded_amount   = xrr.refund_amount
         where xrr.refund_id = v_refund_id;
      end if;
      commit;
      x_status_code := 'S';
      x_status_msg  := 'Refund request has been updated successfully.';
    
    else
      x_status_code := 'F';
      x_status_msg  := 'Error getting the refund ID record';
    
    end if;
  exception
    when others then
      x_status_code := 'F';
      v_error       := sqlerrm;
      x_status_msg  := 'Unexpected Error: ' || substr(v_error, 1, 2000);
  end update_xrr_processing_status;
  procedure update_xrr_Sadad_status(P_SADAD_REFUND_ID           NUMBER,
                                    p_refund_status             varchar2,
                                    p_refund_date               varchar2,
                                    p_RES_ADDITIONAL_ATTRIBUTE1 varchar2,
                                    p_RES_ADDITIONAL_ATTRIBUTE2 varchar2) is
    v_refund_date date;
  begin
    insert into xx_refund_request_param_log
      (P_SADAD_REFUND_ID,
       P_REFUND_STATUS,
       P_REFUND_DATE,
       p_RES_ADDITIONAL_ATTRIBUTE1,
       p_RES_ADDITIONAL_ATTRIBUTE2,
       PARAM_SOURCE,
       CREATION_DATE)
    values
      (P_SADAD_REFUND_ID,
       P_REFUND_STATUS,
       P_REFUND_DATE,
       p_RES_ADDITIONAL_ATTRIBUTE1,
       p_RES_ADDITIONAL_ATTRIBUTE2,
       'RMS',
       sysdate
       
       );
    commit;
  
    v_refund_date := to_date(replace(p_refund_date, 'T', ' '),
                             'YYYY-MM-DD HH24:MI:SS');
    update xx_refund_request xrr
       set xrr.refund_status             = p_refund_status,
           xrr.refund_date               = v_refund_date,
           xrr.res_additional_attribute1 = p_RES_ADDITIONAL_ATTRIBUTE1,
           xrr.res_additional_attribute2 = p_RES_ADDITIONAL_ATTRIBUTE2,
           xrr.manual_processing_flag = case
                                          when p_refund_status = 'Expired' then
                                           'Y'
                                          else
                                           'N'
                                        end,
           refund_method                 = refund_channel
     where xrr.sadad_refund_id = P_SADAD_REFUND_ID;
    commit;
  end;
  procedure update_gl_accounting(p_Prefund_method             VARCHAR2,
                                 p_PRES_ADDITIONAL_ATTRIBUTE1 VARCHAR2,
                                 p_PRES_ADDITIONAL_ATTRIBUTE2 VARCHAR2) is
  begin
    update xx_refund_request
       set refund_method = p_Prefund_method,
       RES_ADDITIONAL_ATTRIBUTE2=p_PRES_ADDITIONAL_ATTRIBUTE2
     where RES_ADDITIONAL_ATTRIBUTE1 = p_PRES_ADDITIONAL_ATTRIBUTE1
       and REFUND_STATUS = 'Processed';
    commit;
  end;
end xx_refund_request_pkg;
